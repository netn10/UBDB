- Search must never 500 on bad input: unknown/malformed operators become `warnings[]`, the rest of the query still runs.
- `page_size` caps at 175 (Scryfall's max); default 60.

```python
def _images(raw: dict) -> dict:
    """Return a card's image_uris, falling back to the front face for DFCs."""
    imgs = raw.get("image_uris")
    if not imgs:
        faces = raw.get("card_faces") or []
        if faces and faces[0].get("image_uris"):
            imgs = faces[0]["image_uris"]
    return imgs or {}


def normalize_print(raw: dict) -> dict:
    """Map one Scryfall printing to a UBDB print record."""
    imgs = _images(raw)
    normal = imgs.get("normal") or imgs.get("large") or imgs.get("png")
    return {
        "scryfall_id": raw.get("id"),
        "set": raw.get("set"),
        "set_name": raw.get("set_name"),
        "collector_number": raw.get("collector_number"),
        "rarity": raw.get("rarity"),
        "released_at": raw.get("released_at"),
        "image_small": imgs.get("small"),
        "image_normal": normal,
        "image_art_crop": imgs.get("art_crop"),
        "image_png": imgs.get("png"),
        "art_uri": normal,
    }


def _face_or_top(raw: dict, key):
    """Prefer a top-level attribute; fall back to the front face (DFCs)."""
    if raw.get(key) is not None:
        return raw.get(key)
    faces = raw.get("card_faces") or []
    front = faces[0] if faces else {}
    return front.get(key)


def group_prints(raw_cards: list) -> list:
    """Group Scryfall printings by oracle_id into ub_card records."""
    by_oracle = {}
    for raw in raw_cards:
        oid = raw.get("oracle_id")
        if not oid:
            continue
        card = by_oracle.get(oid)
        if card is None:
            colors = _face_or_top(raw, "colors")
            card = by_oracle[oid] = {
                "oracle_id": oid,
                "name": raw.get("name"),
                "oracle_text": raw.get("oracle_text", ""),
                "mana_cost": _face_or_top(raw, "mana_cost") or "",
                "type_line": raw.get("type_line", ""),
                "colors": colors if colors is not None else [],
                "color_identity": raw.get("color_identity", []),
                "cmc": raw.get("cmc", 0.0),
                "power": _face_or_top(raw, "power"),
                "toughness": _face_or_top(raw, "toughness"),
                "loyalty": _face_or_top(raw, "loyalty"),
                "keywords": raw.get("keywords", []),
                "layout": raw.get("layout"),
                "rarity": raw.get("rarity"),
                "official_uw_image": None,
                "prints": [],
            }
        card["prints"].append(normalize_print(raw))

    result = []
    for card in by_oracle.values():
        prints = card["prints"]
        card["ub_franchises"] = sorted({p["set_name"] for p in prints if p["set_name"]})
        card["art_uri"] = next((p["art_uri"] for p in prints if p["art_uri"]), None)
        dates = [p["released_at"] for p in prints if p.get("released_at")]
        card["released_at"] = min(dates) if dates else None
        result.append(card)
    return result
```

```python
"""Scryfall-style query parser for UBDB. Parses a query string into predicates,
applies them (implicit AND) to enriched card dicts, then sorts and paginates.
"""
import re

RARITY_RANK = {"common": 0, "uncommon": 1, "rare": 2, "mythic": 3,
               "special": 4, "bonus": 5}
COLOR_LETTERS = set("WUBRG")
OP_RE = re.compile(r"^([a-zA-Z]+)(>=|<=|!=|=|:|>|<)(.*)$")
NUMERIC_KEYS = {"cmc", "mv", "pow", "tou", "loy"}


def tokenize(q):
    """Split a query into (negated, body) tokens, honoring "double quotes"."""
    tokens, i, n = [], 0, len(q or "")
    while i < n:
        if q[i].isspace():
            i += 1
            continue
        neg = False
        if q[i] == "-":
            neg = True
            i += 1
        buf, in_quote = [], False
        while i < n and (in_quote or not q[i].isspace()):
            ch = q[i]
            if ch == '"':
                in_quote = not in_quote
            else:
                buf.append(ch)
            i += 1
        body = "".join(buf)
        if body:
            tokens.append((neg, body))
    return tokens


def _color_set(value):
    v = value.lower()
    if v in ("c", "colorless"):
        return frozenset()
    return frozenset(ch.upper() for ch in v if ch.upper() in COLOR_LETTERS)


def _color_pred(card_key, op, value):
    query = _color_set(value)

    def pred(card):
        have = frozenset(c.upper() for c in (card.get(card_key) or []))
        if op in (":", ">="):
            return query <= have
        if op == "=":
            return have == query
        if op == "<=":
            return have <= query
        if op == ">":
            return query < have
        if op == "<":
            return have < query
        return False
    return pred


def _num(value):
    try:
        return float(value)
    except (TypeError, ValueError):
        return None


def _numeric_pred(card_key, op, value):
    target = _num(value)

    def pred(card):
        actual = _num(card.get(card_key))
        if actual is None or target is None:
            return False
        if op in (":", "="):
            return actual == target
        if op == "!=":
            return actual != target
        if op == ">=":
            return actual >= target
        if op == "<=":
            return actual <= target
        if op == ">":
            return actual > target
        if op == "<":
            return actual < target
        return False
    return pred


def _rarity_pred(op, value):
    want = value.lower()
    want_rank = RARITY_RANK.get(want)

    def pred(card):
        have = (card.get("rarity") or "").lower()
        if op in (":", "="):
            return have == want
        if want_rank is None:
            return False
        have_rank = RARITY_RANK.get(have)
        if have_rank is None:
            return False
        if op == ">=":
            return have_rank >= want_rank
        if op == "<=":
            return have_rank <= want_rank
        if op == ">":
            return have_rank > want_rank
        if op == "<":
            return have_rank < want_rank
        return False
    return pred


def _substr_pred(card_key, value):
    needle = value.lower()
    return lambda card: needle in (card.get(card_key) or "").lower()


def _name_pred(value):
    return _substr_pred("name", value)


def _set_pred(value):
    want = value.lower()
    return lambda card: any((p.get("set") or "").lower() == want
                            for p in card.get("prints", []))


def _franchise_pred(value):
    needle = value.lower()
    return lambda card: any(needle in (f or "").lower()
                            for f in card.get("ub_franchises", []))


def _reskin_pred(reskin_counts):
    return lambda card: reskin_counts.get(card.get("oracle_id"), 0) > 0


def _build(key, op, value, reskin_counts):
    """Return (predicate, warning). Exactly one is non-None."""
    key = key.lower()
    if not value and key not in ("is", "has"):
        return None, f"empty value for operator: {key}:"
    if key in ("t", "type"):
        return _substr_pred("type_line", value), None
    if key in ("o", "oracle"):
        return _substr_pred("oracle_text", value), None
    if key in ("c", "color"):
        return _color_pred("colors", op, value), None
    if key in ("id", "identity"):
        return _color_pred("color_identity", op, value), None
    if key in ("cmc", "mv"):
        return _numeric_pred("cmc", op, value), None
    if key == "pow":
        return _numeric_pred("power", op, value), None
    if key == "tou":
        return _numeric_pred("toughness", op, value), None
    if key == "loy":
        return _numeric_pred("loyalty", op, value), None
    if key in ("r", "rarity"):
        return _rarity_pred(op, value), None
    if key in ("set", "e"):
        return _set_pred(value), None
    if key in ("fr", "franchise"):
        return _franchise_pred(value), None
    if key in ("is", "has"):
        if value.lower() in ("reskinned", "reskin"):
            return _reskin_pred(reskin_counts), None
        return None, f"unknown value: {key}:{value}"
    return None, f"unknown operator: {key}:"


def parse_query(q, reskin_counts=None):
    reskin_counts = reskin_counts or {}
    preds, warnings = [], []
    for neg, body in tokenize(q):
        m = OP_RE.match(body)
        if m:
            pred, warn = _build(m.group(1), m.group(2), m.group(3), reskin_counts)
        else:
            pred, warn = _name_pred(body), None
        if warn:
            warnings.append(warn)
            continue
        preds.append((lambda c, p=pred: not p(c)) if neg else pred)
    return preds, warnings


def _sort_key(order):
    if order == "cmc":
        return lambda c: (c.get("cmc") or 0.0, (c.get("name") or "").lower())
    if order == "rarity":
        return lambda c: (RARITY_RANK.get(c.get("rarity"), -1), (c.get("name") or "").lower())
    if order == "released":
        return lambda c: (c.get("released_at") or "", (c.get("name") or "").lower())
    if order == "franchise":
        return lambda c: ((c.get("ub_franchises") or [""])[0], (c.get("name") or "").lower())
    return lambda c: (c.get("name") or "").lower()


def search(cards, q="", order="name", direction="asc", page=1, page_size=60,
           reskin_counts=None):
    preds, warnings = parse_query(q, reskin_counts)
    matched = [c for c in cards if all(p(c) for p in preds)]
    matched.sort(key=_sort_key(order), reverse=(direction == "desc"))
    page = max(1, page)
    page_size = max(1, min(page_size, 175))
    total = len(matched)
    start = (page - 1) * page_size
    window = matched[start:start + page_size]
    return {
        "cards": window,
        "total": total,
        "page": page,
        "page_size": page_size,
        "has_more": start + page_size < total,
        "warnings": warnings,
    }
```

```python
import random as _random


@app.get("/api/franchises")
def list_franchises():
    counts = {}
    for c in _CARDS:
        for name in c.get("ub_franchises", []):
            counts[name] = counts.get(name, 0) + 1
    franchises = [{"name": n, "count": counts[n]} for n in sorted(counts)]
    return jsonify({"franchises": franchises})


@app.get("/api/random")
def random_card():
    if not _CARDS:
        return jsonify({"error": "no cards"}), 404
    return jsonify({"oracle_id": _random.choice(_CARDS)["oracle_id"]})
```

```tsx
"use client";
import { useState } from "react";
import { useRouter } from "next/navigation";

const COLORS = ["W", "U", "B", "R", "G"];

export default function AdvancedPage() {
  const router = useRouter();
  const [name, setName] = useState("");
  const [type, setType] = useState("");
  const [oracle, setOracle] = useState("");
  const [colors, setColors] = useState<string[]>([]);
  const [mvOp, setMvOp] = useState("<=");
  const [mv, setMv] = useState("");
  const [rarity, setRarity] = useState("");
  const [franchise, setFranchise] = useState("");
  const [reskinned, setReskinned] = useState(false);

  function build(e: React.FormEvent) {
    e.preventDefault();
    const parts: string[] = [];
    if (name) parts.push(name.trim());
    if (type) parts.push(`t:${type.trim()}`);
    if (oracle) parts.push(`o:${oracle.trim()}`);
    if (colors.length) parts.push(`c:${colors.join("").toLowerCase()}`);
    if (mv) parts.push(`cmc${mvOp}${mv.trim()}`);
    if (rarity) parts.push(`r:${rarity}`);
    if (franchise) parts.push(`fr:"${franchise.trim()}"`);
    if (reskinned) parts.push("is:reskinned");
    router.push(`/?q=${encodeURIComponent(parts.join(" "))}`);
  }

  const field = "w-full rounded-lg border border-black/15 bg-canvas px-3 py-2 text-sm dark:border-white/15 dark:bg-canvas-dark";

  return (
    <main className="py-6">
      <h1 className="mb-4 text-2xl font-bold">Advanced search</h1>
      <form onSubmit={build} className="grid max-w-lg gap-4">
        <input className={field} placeholder="Card name" value={name} onChange={(e) => setName(e.target.value)} />
        <input className={field} placeholder="Type line (t:)" value={type} onChange={(e) => setType(e.target.value)} />
        <input className={field} placeholder="Oracle text (o:)" value={oracle} onChange={(e) => setOracle(e.target.value)} />
        <div className="flex gap-2">
          {COLORS.map((c) => (
            <button type="button" key={c}
              onClick={() => setColors((cs) => cs.includes(c) ? cs.filter((x) => x !== c) : [...cs, c])}
              className={`h-9 w-9 rounded-full text-sm font-bold ${colors.includes(c) ? "bg-accent text-white dark:bg-accent-dark" : "border border-black/15 dark:border-white/15"}`}>
              {c}
            </button>
          ))}
        </div>
        <div className="flex gap-2">
          <select className={`${field} w-24`} value={mvOp} onChange={(e) => setMvOp(e.target.value)}>
            {["<=", "=", ">="].map((o) => <option key={o}>{o}</option>)}
          </select>
          <input className={field} placeholder="Mana value" value={mv} onChange={(e) => setMv(e.target.value)} inputMode="numeric" />
        </div>
        <select className={field} value={rarity} onChange={(e) => setRarity(e.target.value)}>
          <option value="">Any rarity</option>
          {["common", "uncommon", "rare", "mythic"].map((r) => <option key={r} value={r}>{r}</option>)}
        </select>
        <input className={field} placeholder="Franchise (fr:)" value={franchise} onChange={(e) => setFranchise(e.target.value)} />
        <label className="flex items-center gap-2 text-sm">
          <input type="checkbox" checked={reskinned} onChange={(e) => setReskinned(e.target.checked)} />
          Only cards with a reskin
        </label>
        <button className="rounded-lg bg-accent px-4 py-2 text-white dark:bg-accent-dark">Search</button>
      </form>
```
