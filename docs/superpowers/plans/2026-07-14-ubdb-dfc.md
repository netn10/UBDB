# UBDB Double-Faced Card Handling Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Preview a double-faced card's back on hover in the grid, show both faces on the card page with per-face reskin sections, and add a setting that expands each DFC into a separate back-face tile.

**Architecture:** Extend the sync to store per-face text (`faces[]`) and per-print back images. The frontend gains a `DfcTile` (CSS 3D hover-flip), a both-sides card page, and a client setting (`localStorage`) that expands DFCs into a second tile. Back-face tiles are a display-only client expansion — search/sort/pagination stay per-card.

**Tech Stack:** Python 3.12 + Flask + pytest (backend); Next.js 14 + React 18 + TypeScript + Tailwind (frontend); Scryfall REST API (sync only).

## Global Constraints

- **No git actions by the implementer.** Do NOT run `git`, commit, push, or deploy. The "Commit" blocks are reference-only for the maintainer; the implementer stops after each task's verify step.
- **No AI-authored comments or mentions** anywhere — code comments read as ordinary human comments.
- Target layouts for two-image DFC behavior are exactly `transform` and `modal_dfc`. All other layouts render as today (single front image, no faces, no flip, no back tile).
- A DFC shares one `oracle_id`; a face is identified by index (`0` = front/single, `1` = back). Reskin `face` defaults to `0` when missing.
- `faces` is `[]` for single-faced cards and length 2 for a two-image DFC.
- Cross-origin images route through the Next `image-proxy` (host `cards.scryfall.io` is already allowlisted); local `/` and `data:` URIs do not.
- `data/ub_cards/cards.json` is read-only + Scryfall-generated; never hand-edit it.
- Setting is client-only (`localStorage` key `ubdb.settings`), default `{ splitDfcTiles: false }`.
- Frontend deliverables are verified by `npm run build` (type-clean) plus a runtime smoke check; the repo has no frontend unit-test runner. Backend uses pytest.

Add these two helpers just above `normalize_print`:

```python
def _back_images(raw: dict) -> dict:
    """Return the back-face image_uris for a two-faced card, else empty."""
    faces = raw.get("card_faces") or []
    if len(faces) > 1:
        return faces[1].get("image_uris") or {}
    return {}


def _build_faces(raw: dict) -> list:
    """Return per-face text for a two-image DFC, else an empty list.

    Only transform / modal_dfc cards with two image-bearing faces are treated
    as double-faced here; every other layout renders single-faced.
    """
    if raw.get("layout") not in ("transform", "modal_dfc"):
        return []
    faces = raw.get("card_faces") or []
    if len(faces) < 2:
        return []
    if not (faces[0].get("image_uris") and faces[1].get("image_uris")):
        return []
    out = []
    for f in faces[:2]:
        out.append({
            "name": f.get("name"),
            "mana_cost": f.get("mana_cost", ""),
            "type_line": f.get("type_line", ""),
            "oracle_text": f.get("oracle_text", ""),
            "colors": f.get("colors", []),
            "power": f.get("power"),
            "toughness": f.get("toughness"),
            "loyalty": f.get("loyalty"),
        })
    return out
```

```tsx
"use client";
import Link from "next/link";
import { useEffect, useRef, useState } from "react";
import { UbCard } from "@/types/types";
import { getImageSrc } from "@/lib/api";

const FLIP_DELAY_MS = 150;
const HOLD_MS = 2000;

export default function DfcTile({ card }: { card: UbCard }) {
  const [flipped, setFlipped] = useState(false);
  const timers = useRef<number[]>([]);
  const front = card.prints[0]?.image_normal ?? card.art_uri;
  const back = card.prints[0]?.image_back_normal ?? null;

  function clearTimers() {
    timers.current.forEach((t) => clearTimeout(t));
    timers.current = [];
  }
  useEffect(() => clearTimers, []);

  function onEnter() {
    if (!back) return;
    if (window.matchMedia?.("(prefers-reduced-motion: reduce)").matches) return;
    clearTimers();
    timers.current.push(window.setTimeout(() => setFlipped(true), FLIP_DELAY_MS));
    timers.current.push(window.setTimeout(() => setFlipped(false), FLIP_DELAY_MS + HOLD_MS));
  }
  function onLeave() {
    clearTimers();
    setFlipped(false);
  }

  return (
    <Link href={`/card/${card.oracle_id}`} onMouseEnter={onEnter} onMouseLeave={onLeave}
          className="block [perspective:1000px]">
      <div className={`relative aspect-[5/7] w-full transition-transform duration-500 [transform-style:preserve-3d] ${flipped ? "[transform:rotateY(180deg)]" : ""}`}>
        {front && (
          <img src={getImageSrc(front)} alt={card.name} loading="lazy"
               className="absolute inset-0 h-full w-full rounded-xl shadow [backface-visibility:hidden]" />
        )}
        {back && (
          <img src={getImageSrc(back)} alt={`${card.name} (back)`} loading="lazy"
               className="absolute inset-0 h-full w-full rounded-xl shadow [transform:rotateY(180deg)] [backface-visibility:hidden]" />
        )}
      </div>
    </Link>
  );
}
```

```ts
"use client";
import { useEffect, useState } from "react";

export interface Settings {
  splitDfcTiles: boolean;
}

const KEY = "ubdb.settings";
const DEFAULT: Settings = { splitDfcTiles: false };
const EVENT = "ubdb-settings";

export function readSettings(): Settings {
  if (typeof window === "undefined") return DEFAULT;
  try {
    return { ...DEFAULT, ...JSON.parse(localStorage.getItem(KEY) || "{}") };
  } catch {
    return DEFAULT;
  }
}

export function writeSettings(next: Settings): void {
  localStorage.setItem(KEY, JSON.stringify(next));
  window.dispatchEvent(new Event(EVENT));
}

export function useSettings(): Settings {
  const [settings, setSettings] = useState<Settings>(DEFAULT);
  useEffect(() => {
    const sync = () => setSettings(readSettings());
    sync();
    window.addEventListener(EVENT, sync);
    window.addEventListener("storage", sync);
    return () => {
      window.removeEventListener(EVENT, sync);
      window.removeEventListener("storage", sync);
    };
  }, []);
  return settings;
}
```

- [ ] **Step 4: Expand DFCs in `CardGrid`** — replace `src/components/CardGrid.tsx` (this makes it a client component and adds the back tile):

```tsx
"use client";
import Link from "next/link";
import { Fragment } from "react";
import { UbCard } from "@/types/types";
import { getImageSrc } from "@/lib/api";
import { useSettings } from "@/lib/settings";
import DfcTile from "./DfcTile";

function isFlipDfc(c: UbCard): boolean {
  return c.faces.length === 2 && !!c.prints[0]?.image_back_normal;
}

function BackTile({ card }: { card: UbCard }) {
  const back = card.prints[0]!.image_back_normal!;
  return (
    <Link href={`/card/${card.oracle_id}#face-back`} className="group block">
      <img src={getImageSrc(back)} alt={`${card.faces[1].name} (back)`} loading="lazy"
           className="w-full rounded-xl shadow transition group-hover:-translate-y-1" />
    </Link>
  );
}

function PlainTile({ card }: { card: UbCard }) {
  const img = card.prints[0]?.image_normal ?? card.art_uri;
  return (
    <Link href={`/card/${card.oracle_id}`} className="group block">
      {img ? (
        <img src={getImageSrc(img)} alt={card.name} loading="lazy"
             className="w-full rounded-xl shadow transition group-hover:-translate-y-1" />
      ) : (
        <div className="flex aspect-[5/7] items-center justify-center rounded-xl bg-black/10 p-2 text-center text-xs dark:bg-white/10">
          {card.name}
        </div>
      )}
    </Link>
  );
}

export default function CardGrid({ cards }: { cards: UbCard[] }) {
  const { splitDfcTiles } = useSettings();
  return (
    <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5">
      {cards.map((c) =>
        isFlipDfc(c) ? (
          <Fragment key={c.oracle_id}>
            <DfcTile card={c} />
            {splitDfcTiles && <BackTile card={c} />}
          </Fragment>
        ) : (
          <PlainTile key={c.oracle_id} card={c} />
        )
      )}
    </div>
  );
}
```
